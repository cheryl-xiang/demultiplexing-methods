#To Run: Paste entire script into terminal

conda activate demuxtags

#mcginnis_ms
rm -f results/demuxtags/mcginnis_ms/mcginnis_ms.counts
mkdir -p results/demuxtags/mcginnis_ms
{ time methods/demuxtags/demuxtags_files/demux_tags \
    -B data/mcginnis_ms/hto/mex/barcodes.tsv \
    -F data/mcginnis_ms/hto/mex/features.tsv \
    -M data/mcginnis_ms/hto/mex/matrix.mtx \
    -o results/demuxtags/mcginnis_ms/mcginnis_ms ; } 2>&1 | tee results/demuxtags/mcginnis_ms/time.txt
